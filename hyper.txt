-- Instance oluşturucu
function UILib.Create(className, props, children)
    local inst = Instance.new(className)
    for k, v in pairs(props or {}) do inst[k] = v end
    for _, child in ipairs(children or {}) do child.Parent = inst end
    return inst
end
local Create = UILib.Create

-- Sürüklenebilir yapma
function UILib.MakeDraggable(frame, handle)
    local dragging, dragInput, mousePos, framePos = false, nil, nil, nil
    handle = handle or frame

    handle.InputBegan:Connect(function(inp)
        if inp.UserInputType == Enum.UserInputType.MouseButton1
        or inp.UserInputType == Enum.UserInputType.Touch then
            dragging  = true
            mousePos  = inp.Position
            framePos  = frame.Position
            inp.Changed:Connect(function()
                if inp.UserInputState == Enum.UserInputState.End then
                    dragging = false
                end
            end)
        end
    end)

    handle.InputChanged:Connect(function(inp)
        if inp.UserInputType == Enum.UserInputType.MouseMovement
        or inp.UserInputType == Enum.UserInputType.Touch then
            dragInput = inp
        end
    end)

    UserInputService.InputChanged:Connect(function(inp)
        if not dragging or inp ~= dragInput then return end
        local delta = inp.Position - mousePos
        frame.Position = UDim2.new(
            framePos.X.Scale, framePos.X.Offset + delta.X,
            framePos.Y.Scale, framePos.Y.Offset + delta.Y
        )
    end)
end
local MakeDraggable = UILib.MakeDraggable

    task.spawn(function()

            "Connecting to services...", "Optimizing performance...",

                if v:IsA("Frame") or v:IsA("TextLabel") or v:IsA("ImageLabel") then
                    TweenService:Create(v, TweenInfo.new(0.6), {
                        BackgroundTransparency = 1, TextTransparency = 1
                    }):Play()
                end
            end)
        end
        TweenService:Create(LoadScreen, TweenInfo.new(0.6), { BackgroundTransparency = 1 }):Play()
        task.wait(0.7)
        LoadGui:Destroy()
    end)

        if CurrentPage == pageName then return end
        local oldPage = Pages[CurrentPage]
        local newPage = Pages[pageName]
        if not newPage then return end

        local inLayout = newPage:FindFirstChildWhichIsA("UIListLayout")
        if inLayout then
            newPage.CanvasSize = UDim2.new(0,0,0, inLayout.AbsoluteContentSize.Y + 20)
        end

        newPage.Position = UDim2.new(1,0,0,0)
        newPage.Visible  = true

        TweenService:Create(oldPage, TweenInfo.new(0.3, Enum.EasingStyle.Quint), {
            Position = UDim2.new(-1,0,0,0)
        }):Play()

        local tw = TweenService:Create(newPage, TweenInfo.new(0.3, Enum.EasingStyle.Quint), {
            Position = UDim2.new(0,0,0,0)
        })
        tw:Play()
        tw.Completed:Connect(function()
            oldPage.Visible  = false
            oldPage.Position = UDim2.new(0,0,0,0)
        end)

        CurrentPage = pageName
    end

            if CurrentPage ~= data.Name then
                TweenService:Create(tabBtn, TweenInfo.new(0.2), { BackgroundColor3 = Colors.SurfaceLight }):Play()
            end
        end)
        tabBtn.MouseLeave:Connect(function()
            if CurrentPage ~= data.Name then
                TweenService:Create(tabBtn, TweenInfo.new(0.2), { BackgroundColor3 = Colors.Surface }):Play()
            end

            for _, b in ipairs(TabButtons) do
                TweenService:Create(b, TweenInfo.new(0.2), {
                    BackgroundColor3 = Colors.Surface,
                    TextColor3 = Colors.TextSecondary

                BackgroundColor3 = Colors.Accent1,
                TextColor3 = Colors.TextPrimary

            if glow then
                TweenService:Create(glow, TweenInfo.new(0.2), { Transparency = 0.3 }):Play()
            end

        end)

        table.insert(TabButtons, tabBtn)

        local tabPage = Create("ScrollingFrame", {
            Name = data.Name .. "Page",
            Size = UDim2.new(1,0,1,0),
            Position = UDim2.new(0,0,0,0),

            Padding = UDim.new(0,10),
            SortOrder = Enum.SortOrder.LayoutOrder,
            Parent = tabPage
        })

            tabPage.CanvasSize = UDim2.new(0,0,0, tabLayout.AbsoluteContentSize.Y + 20)
        end)

        Pages[data.Name] = tabPage
    end

            Create("UICorner", { CornerRadius = UDim.new(0,10) }),
            Create("UIStroke", { Color = Colors.Accent2, Transparency = 0.5 })
        })

        Create("TextLabel", {
            Text = title,
            Size = UDim2.new(1,-80,1,0),

    -- Slider
    local function CreateSlider(parent, title, min, max, default, suffix, callback)
        ToggleStates[title] = default
        if callback then SettingsCallbacks[title] = callback end

        local container = Create("Frame", {
            Size = UDim2.new(1,0,0,60),
            BackgroundColor3 = Colors.Surface,
            Parent = parent
        }, {
            Create("UICorner", { CornerRadius = UDim.new(0,10) }),
            Create("UIStroke", { Color = Colors.Accent2, Transparency = 0.5 })
        })

        Create("TextLabel", {
            Text = title,
            Size = UDim2.new(1,-20,0,20),
            Position = UDim2.new(0,15,0,8),
            BackgroundTransparency = 1,
            Font = Enum.Font.Gotham,
            TextSize = 14,
            TextColor3 = Colors.TextPrimary,
            TextXAlignment = Enum.TextXAlignment.Left,
            Parent = container
        })

        local valueLabel = Create("TextLabel", {
            Text = tostring(default) .. (suffix or ""),
            Size = UDim2.new(0,60,0,20),
            Position = UDim2.new(1,-75,0,8),
            BackgroundTransparency = 1,
            Font = Enum.Font.GothamBold,
            TextSize = 14,
            TextColor3 = Colors.Accent1,
            TextXAlignment = Enum.TextXAlignment.Right,
            Parent = container
        })

        local track = Create("Frame", {
            Size = UDim2.new(1,-40,0,8),
            Position = UDim2.new(0,20,0,38),
            BackgroundColor3 = Colors.SurfaceLight,
            Parent = container
        }, { Create("UICorner", { CornerRadius = UDim.new(0,4) }) })

        local fill = Create("Frame", {
            Size = UDim2.new((default-min)/(max-min),0,1,0),
            BackgroundColor3 = Colors.Accent1,
            Parent = track
        }, {
            Create("UICorner", { CornerRadius = UDim.new(0,4) }),
            Create("UIGradient", {
                Color = ColorSequence.new({
                    ColorSequenceKeypoint.new(0, Colors.Accent1),
                    ColorSequenceKeypoint.new(1, Colors.Accent3)
                })
            })
        })

        Create("Frame", {
            Size = UDim2.new(0,20,0,20),
            Position = UDim2.new(1,-10,0.5,-10),
            BackgroundColor3 = Colors.Accent1,
            Parent = fill
        }, {
            Create("UICorner", { CornerRadius = UDim.new(1,0) }),
            Create("UIStroke", { Color = Colors.TextPrimary, Thickness = 2 })
        })

        local dragging = false

        local function Update(input)
            local x   = math.clamp(input.Position.X - track.AbsolutePosition.X, 0, track.AbsoluteSize.X)
            local pct = x / track.AbsoluteSize.X
            fill.Size = UDim2.new(pct,0,1,0)
            local val = math.floor((min + (max-min)*pct) * 10) / 10
            valueLabel.Text = tostring(val) .. (suffix or "")
            ToggleStates[title] = val
            if callback then callback(val) end
        end

        track.InputBegan:Connect(function(input)
            if input.UserInputType == Enum.UserInputType.MouseButton1 then
                dragging = true; Update(input)
            end
        end)
        UserInputService.InputChanged:Connect(function(input)
            if dragging and input.UserInputType == Enum.UserInputType.MouseMovement then
                Update(input)
            end
        end)
        UserInputService.InputEnded:Connect(function(input)
            if input.UserInputType == Enum.UserInputType.MouseButton1 then
                dragging = false
            end
        end)

        return container
    end

    -- Ayarları kaydet/yükle
    local SETTINGS_FILE = "HyperHub_Settings.json"
    local HttpSvc = game:GetService("HttpService")

    local function SaveSettings()
        if not writefile then
            UILib.Notify("Settings", "Executor writefile desteği yok.", 3) return
        end
        local data = {}
        for k,v in pairs(ToggleStates) do data[k] = v end
        local ok, err = pcall(function() writefile(SETTINGS_FILE, HttpSvc:JSONEncode(data)) end)
        if ok then UILib.Notify("✅ Kaydedildi", "Tüm ayarlar kaydedildi.", 3)
        else UILib.Notify("❌ Hata", tostring(err):sub(1,40), 4) end
    end

    local function LoadSettings()
        if not readfile or not isfile then
            UILib.Notify("Settings", "Executor readfile desteği yok.", 3) return
        end
        if not isfile(SETTINGS_FILE) then
            UILib.Notify("Settings", "Kayıtlı ayar bulunamadı.", 3) return
        end
        local ok, data = pcall(function() return HttpSvc:JSONDecode(readfile(SETTINGS_FILE)) end)
        if not ok or type(data) ~= "table" then
            UILib.Notify("❌ Hata", "Ayar dosyası bozuk.", 3) return
        end
        local loaded = 0
        for title, value in pairs(data) do
            local cb = SettingsCallbacks[title]
            if cb then
                ToggleStates[title] = value
                pcall(cb, value)
                loaded = loaded + 1
            end
        end
        UILib.Notify("✅ Yüklendi", loaded .. " ayar geri yüklendi.", 3)
    end

    -- Klavye kısayolu: RightControl → pencereyi aç/kapat
    UserInputService.InputBegan:Connect(function(inp, gpe)
        if gpe then return end
        if inp.KeyCode == Enum.KeyCode.RightControl then
            MainPanel.Visible = not MainPanel.Visible
        end
    end)

        MainPanel.Visible = true
    end)

        end,
        MainPanel   = MainPanel,
        HubGui      = HubGui,
        Pages       = Pages,
        ToggleStates = ToggleStates,
    }
